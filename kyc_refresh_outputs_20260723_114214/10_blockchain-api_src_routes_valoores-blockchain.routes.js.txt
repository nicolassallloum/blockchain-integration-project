   352	 * GET /api/v1/valoores-blockchain/kyc-daily-created?month=2026-06
   353	 *
   354	 * Purpose:
   355	 * Display number of KYC records created on a daily basis by selected month.
   356	 */
   357	router.get('/kyc-daily-created', async (req, res) => {
   358	  try {
   359	    const month = req.query.month || new Date().toISOString().slice(0, 7);
   360	
   361	    if (!/^\d{4}-\d{2}$/.test(month)) {
   362	      return res.status(400).json({
   363	        success: false,
   364	        message: 'Invalid month format. Use YYYY-MM, example: 2026-06'
   365	      });
   366	    }
   367	
   368	    const dateFrom = `${month}-01`;
   369	
   370	    const result = await pool.query(
   371	      `
   372	      WITH month_days AS (
   373	        SELECT generate_series(
   374	          date_trunc('month', $1::date),
   375	          date_trunc('month', $1::date) + interval '1 month' - interval '1 day',
   376	          interval '1 day'
   377	        )::date AS kyc_date
   378	      ),
   379	      daily_counts AS (
   380	        SELECT
   381	          created_at::date AS kyc_date,
   382	          COUNT(*)::int AS total_kyc_created,
   383	          COUNT(*) FILTER (WHERE blockchain_status = 'CONFIRMED')::int AS confirmed_kyc,
   384	          COUNT(*) FILTER (WHERE blockchain_status = 'FAILED')::int AS failed_kyc,
   385	          COUNT(*) FILTER (WHERE blockchain_status = 'PENDING')::int AS pending_kyc
   386	        FROM blockchain.valoores_customer_blockchain_proofs
   387	        WHERE created_at >= date_trunc('month', $1::date)
   388	          AND created_at < date_trunc('month', $1::date) + interval '1 month'
   389	        GROUP BY created_at::date
   390	      )
   391	      SELECT
   392	        to_char(md.kyc_date, 'YYYY-MM-DD') AS kyc_date,
   393	        COALESCE(dc.total_kyc_created, 0)::int AS total_kyc_created,
   394	        COALESCE(dc.confirmed_kyc, 0)::int AS confirmed_kyc,
   395	        COALESCE(dc.failed_kyc, 0)::int AS failed_kyc,
   396	        COALESCE(dc.pending_kyc, 0)::int AS pending_kyc
   397	      FROM month_days md
   398	      LEFT JOIN daily_counts dc ON dc.kyc_date = md.kyc_date
   399	      ORDER BY md.kyc_date;
   400	      `,
   401	      [dateFrom]
   402	    );
   403	
   404	    const summaryResult = await pool.query(
   405	      `
   406	      SELECT
   407	        COUNT(*)::int AS total_kyc_created,
   408	        COUNT(*) FILTER (WHERE blockchain_status = 'CONFIRMED')::int AS confirmed_kyc,
   409	        COUNT(*) FILTER (WHERE blockchain_status = 'FAILED')::int AS failed_kyc,
   410	        COUNT(*) FILTER (WHERE blockchain_status = 'PENDING')::int AS pending_kyc
   411	      FROM blockchain.valoores_customer_blockchain_proofs
   412	      WHERE created_at >= date_trunc('month', $1::date)
   413	        AND created_at < date_trunc('month', $1::date) + interval '1 month';
   414	      `,
   415	      [dateFrom]
   416	    );
   417	
   418	    return res.status(200).json({
   419	      success: true,
   420	      message: 'KYC daily created report retrieved successfully',
   421	      data: {
   422	        month,
   423	        summary: summaryResult.rows[0],
   424	        daily: result.rows
   425	      }
   426	    });
   427	  } catch (error) {
   428	    console.error('[VALOORES KYC DAILY CREATED ERROR]', error);
   429	    return res.status(500).json({
   430	      success: false,
   431	      message: 'Failed to retrieve KYC daily created report',
   432	      error: error.message
   433	    });
   434	  }
   435	});
