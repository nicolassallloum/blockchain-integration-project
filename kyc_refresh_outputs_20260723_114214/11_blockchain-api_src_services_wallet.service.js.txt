   390	
   391	  const insertColumns = [];
   392	  const insertValues = [];
   393	  const placeholders = [];
   394

   402

   407	  `;
   408

   663	  );
   664	
   665	  if (existingOrgWallet.rowCount > 0) {
   666	    throw new Error(
   667	      `Organization wallet already exists for organizationId: ${resolvedOrganizationId}`
   668	    );
   669	  }
   670	
   671	  const initialBalance = normalizeNumber(
   672	    payload.initialBalance ?? payload.currentBalance ?? payload.current_balance,
   673	    0
   674	  );
   675	
   676	  if (initialBalance < 0) {
   677	    throw new Error('initialBalance must be zero or greater');
   678	  }
   679	
   680	  const plainPassword =
   681	    payload.passwordHash ||
   682	    payload.password ||
   683	    generateOneTimePassword('Org');
   684	
   685	  const normalizedPasswordHash = String(plainPassword).startsWith('$2')
   686	    ? String(plainPassword)
   687	    : await bcrypt.hash(String(plainPassword), 10);
   688	
   689	  const customerId = `ORG_${String(resolvedOrganizationId)
   690	    .replace(/-/g, '')
   691	    .slice(0, 24)}`;
   692	
   693	  const organizationName = organization.organization_name;
   694	  const emailHash = buildOrganizationEmail(organizationName);
   695	
   696	  let fabricResult;
   697	  let blockchainWallet;
   698	  let blockchainTransaction;
   699	
   700	  try {
   701	    fabricResult = await fabricService.submitTransaction(
   702	      'CreateWallet',
   703	      [
   704	        customerId,
   705	
   706	        /*
   707	         * IMPORTANT:
   708	         * Organization wallet also sends organization_code to Fabric.
   709	         */
   710	        organizationCode,
   711	
   712	        organizationName,
   713	        organizationCode,
   714	        '',
   715	        emailHash,
   716	        normalizedPasswordHash,
   717	        String(initialBalance)
   718	      ],
   719	      {
   720	        requestId: payload.requestId || payload.request_id || null,
   721	        correlationId: payload.correlationId || payload.correlation_id || null,
   722	        sourceSystem: payload.sourceSystem || payload.source_system || 'BLOCKCHAIN_API',
   723	        requestSource: payload.requestSource || payload.request_source || 'ANGULAR_TEST_UI',
   724	        createdBy: payload.createdBy || payload.created_by || 'angular-test-ui'
   725	      }
   726	    );
   727	
   728	    const extracted = extractBlockchainWallet(fabricResult);
   729	    blockchainWallet = extracted.wallet;
   730	    blockchainTransaction = extracted.transaction;
   731	  } catch (error) {
   732	    throw new Error(`Blockchain organization wallet creation failed: ${error.message}`);
   733	  }
   734	
   735	  const walletAddress = blockchainWallet.walletAddress;
   736	  const fabricTransactionId = getFabricTransactionId(
   737	    blockchainWallet,
   738	    blockchainTransaction
   739	  );
   740	
   741	  const client = await db.getClient();
   742	
   743	  let enterpriseSaveResult;
   744	
   745	  try {
   746	    await client.query('BEGIN');
   747	
   748	    enterpriseSaveResult = await enterprisePersistenceRepository.saveWalletEnterprise(
   749	      client,
   750	      {
   751	        walletAddress,
   752	        customerId,
   753	        organizationId: resolvedOrganizationId,
   754	        organizationCode,
   755	        walletType: 'ORGANIZATION',
   756	        fullName: organizationName,
   757	        nationalIdHash: organizationCode,
   758	        mobileHash: null,
   759	        emailHash,
   760	        passwordHash: normalizedPasswordHash,
   761	        ledgerDocType: 'organization_wallet',
   762	        currentBalance: blockchainWallet.balance ?? initialBalance,
   763	        currencyCode: payload.currencyCode || payload.currency || 'USD',
   764	        status: blockchainWallet.status || 'ACTIVE',
   765	        fabricTxId: fabricTransactionId,
   766	        fabricChannelName: fabricResult.channelName || DEFAULT_CHANNEL_NAME,
   767	        chaincodeName: fabricResult.chaincodeName || DEFAULT_CHAINCODE_NAME,
   768	        walletMetadata: {
   769	          source: 'BLOCKCHAIN_API',
   770	          walletType: 'ORGANIZATION',
   771	          fabricTxId: fabricTransactionId,
   772	          organizationId: resolvedOrganizationId,
   773	          organizationCode
   774	        },
   775	        kycPayload: {
   776	          organizationId: resolvedOrganizationId,
   777	          organizationName,
   778	          organizationCode
   779	        },
   780	        blockchainPayload: blockchainWallet,
   781	        fabricResponse: fabricResult,
   782	        requestId: payload.requestId || payload.request_id || null,
   783	        requestSource: payload.requestSource || payload.request_source || 'ANGULAR_TEST_UI',
   784	        sourceSystem: payload.sourceSystem || payload.source_system || 'BLOCKCHAIN_API',
   785	        createdBy: payload.createdBy || payload.created_by || 'angular-test-ui',
   786	        updatedBy: payload.createdBy || payload.created_by || 'angular-test-ui',
   787	        originalPayload: {
   788	          ...payload,
   789	          organizationId: resolvedOrganizationId,
   790	          organizationCode
   791	        }
   792	      }
   793	    );
   794	
   795	    await client.query('COMMIT');
   796	  } catch (error) {
   797	    await client.query('ROLLBACK');
   798	
   799	    throw new Error(
   800	      `Organization wallet created on Fabric but enterprise PostgreSQL save failed. Manual reconciliation required. FabricTxId=${fabricTransactionId || 'N/A'} WalletAddress=${walletAddress}. Error: ${error.message}`
   801	    );
   802	  } finally {
   803	    client.release();
   804	  }
   805	
   806	  const row = enterpriseSaveResult.wallet;
   807	
   808	  const profile = await getProfessionalWalletProfileByAddress(row.wallet_address);
   809	
   810	  return {
   811	    wallet: profile || row,
   812	    blockchain: {
   813	      saved: true,
   814	      walletAddress,
   815	      fabricTransactionId,
   816	      channelName: fabricResult.channelName || DEFAULT_CHANNEL_NAME,
   817	      chaincodeName: fabricResult.chaincodeName || DEFAULT_CHAINCODE_NAME,
   818	      transaction: blockchainTransaction || null
   819	    },
   820	    postgres: {
   821	      saved: true,
   822	      walletId: row.wallet_id || null
   823	    },
   824	    organization: {
   825	      organizationId: resolvedOrganizationId,
   826	      organizationCode
   827	    },
   828	    oneTimePassword: String(plainPassword).startsWith('$2') ? null : plainPassword
   829	  };
   830	};

   832	exports.loginWallet = async ({ walletAddress, customerId, password }) => {
   833	  const loginIdentifier = walletAddress || customerId;
   834	
   835	  if (!loginIdentifier) {
   836	    throw new Error('walletAddress is required');
   837	  }
   838	
   839	  if (!password) {
   840	    throw new Error('password is required');
   841	  }
   842	
   843	  const result = await db.query(
   844	    `
   845	    SELECT
   846	      wallet_id,
   847	      wallet_address,
   848	      customer_id,
   849	      organization_id,
   850	      organization_code,
   851	      wallet_type,
   852	      full_name,
   853	      password_hash,
   854	      status
   855	    FROM blockchain.wallets
   856	    WHERE wallet_address = $1
   857	    LIMIT 1
   858	    `,
   859	    [loginIdentifier]
   860	  );
   861	
   862	  if (result.rows.length === 0) {
   863	    return null;
   864	  }
   865	
   866	  const wallet = result.rows[0];
   867	
   868	  if (!wallet.password_hash) {
   869	    throw new Error('Wallet has no password hash configured');
   870	  }
   871	
   872	  const isPasswordValid = await bcrypt.compare(password, wallet.password_hash);
   873	
   874	  if (!isPasswordValid) {
   875	    return null;
   876	  }
   877	
   878	  const profile =
   879	    (await getProfessionalWalletProfileByAddress(wallet.wallet_address)) ||
   880	    (await getProfessionalWalletProfileByCustomerId(wallet.customer_id)) ||
   881	    {};
   882	
   883	  const walletType = normalizeWalletType(profile.walletType || wallet.wallet_type);
   884	
   885	  const token = jwt.sign(
   886	    {
   887	      walletId: wallet.wallet_id,
   888	      walletAddress: wallet.wallet_address,
   889	      customerId: wallet.customer_id,
   890	      organizationId: wallet.organization_id,
   891	      organizationCode: wallet.organization_code || null,
   892	      walletType
   893	    },
   894	    JWT_SECRET,
   895	    {
   896	      expiresIn: DEFAULT_JWT_EXPIRES_IN
   897	    }
   898	  );
   899	
   900	  return {
   901	    token,
   902	    wallet: {
   903	      walletId: wallet.wallet_id,
   904	      walletAddress: profile.walletAddress || wallet.wallet_address,
   905	      customerId: profile.customerId || wallet.customer_id,
   906	      organizationId: profile.organizationId || wallet.organization_id,
   907	      organizationName: profile.organizationName || null,
   908	      organizationCode: profile.organizationCode || wallet.organization_code,
   909	      walletType,
   910	      fullName: profile.fullName || wallet.full_name,
   911	      customerName: profile.fullName || wallet.full_name,
   912	      nationalIdHash: profile.nationalIdHash || null,
   913	      countryName: profile.countryName || null,
   914	      emailHash: profile.emailHash || null,
   915	      mobileHash: profile.mobileHash || null,
   916	      currentBalance: profile.currentBalance ?? 0,
   917	      currencyCode: profile.currencyCode || 'TOKEN',
   918	      createdAt: profile.createdAt || null,
   919	      status: profile.status || wallet.status
   920	    }
   921	  };
   922	};
