    52	function getErrorStatusCode(error, defaultStatus = 400) {
    53	  const message = String(error.message || '').toLowerCase();
    54	
    55	  if (
    56	    message.includes('blockchain wallet creation failed') ||
    57	    message.includes('fabric') ||
    58	    message.includes('endorsement') ||
    59	    message.includes('deadline') ||
    60	    message.includes('gateway') ||
    61	    message.includes('grpc')
    62	  ) {
    63	    return 502;
    64	  }
    65	
    66	  if (message.includes('already exists')) {
    67	    return 409;
    68	  }
    69	
    70	  if (message.includes('not found')) {
    71	    return 404;
    72	  }
    73	
    74	  return defaultStatus;
    75	}

    97	 */
    98	exports.listWallets = async (req, res) => {
    99	  try {
   100	    const page = Math.max(parseInt(req.query.page || '1', 10), 1);
   101	    const limit = Math.min(Math.max(parseInt(req.query.limit || '13', 10), 1), 100);
   102	    const offset = (page - 1) * limit;
   103	    const search = (req.query.search || '').trim();
   104	
   105	    const result = await walletService.listWallets({
   106	      page,
   107	      limit,
   108	      offset,
   109	      search
   110	    });
   111	
   112	    return res.status(200).json({
   113	      success: true,
   114	      message: 'Wallet list retrieved successfully',
   115	      data: result.data,
   116	      pagination: {
   117	        page,
   118	        limit,
   119	        totalRecords: result.totalRecords,
   120	        totalPages: Math.ceil(result.totalRecords / limit),
   121	        hasNextPage: page < Math.ceil(result.totalRecords / limit),
   122	        hasPreviousPage: page > 1
   123	      },
   124	      filters: {
   125	        search: search || null
   126	      },
   127	      source: 'postgres',
   128	      requestId: getRequestId(req)
   129	    });
   130	  } catch (error) {
   131	    return res.status(500).json({
   132	      success: false,
   133	      message: 'Failed to retrieve wallet list',
   134	      error: {
   135	        message: error.message,
   136	        stack: process.env.NODE_ENV === 'production' ? undefined : error.stack
   137	      },
   138	      requestId: getRequestId(req)
   139	    });
   140	  }
