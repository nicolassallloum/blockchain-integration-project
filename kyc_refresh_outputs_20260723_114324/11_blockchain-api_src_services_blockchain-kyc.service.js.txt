   169	async function markKycRequestWalletCreated(requestId, walletResult) {
   170	  const walletAddress = extractWalletAddress(walletResult);
   171	  const fabricTxId = extractFabricTxId(walletResult);
   172	
   173	  const result = await db.query(
   174	    `
   175	    UPDATE blockchain.blockchain_kyc_wallet_requests
   176	    SET
   177	      wallet_address = $1,
   178	      ledger_reference = $2,
   179	      blockchain_tx_id = $3,
   180	      request_status = 'WALLET_CREATED',
   181	      error_message = NULL,
   182	      updated_at = NOW()
   183	    WHERE request_id = $4
   184	    RETURNING *
   185	    `,
   186	    [walletAddress, fabricTxId, fabricTxId, requestId]
   187	  );
   188	
   189	  return result.rows[0];
   190	}

   211	async function createBlockchainKycWallet(payload, files) {
   212	  if (!payload.customerId) {
   213	    throw new Error('customerId is required');
   214	  }
   215	
   216	  if (!payload.organizationId) {
   217	    throw new Error('organizationId is required');
   218	  }
   219	
   220	  const kycRequest = await insertKycRequest(payload, files);
   221	
   222	  try {
   223	    const walletResult = await walletService.createWallet({
   224	      ...payload,
   225	
   226	      customerId: payload.customerId,
   227	
   228	      organizationId: payload.organizationId,
   229	      organizationCode: payload.organizationCode,
   230	      organizationName: payload.organization || payload.organizationName,
   231	
   232	      fullName: resolveFullName(payload),
   233	
   234	      nationalIdHash: payload.nationalIdHash || payload.legalIdNumberHash || '',
   235	      mobileHash: payload.mobileHash || '',
   236	      emailHash: payload.emailHash || '',
   237	
   238	      password:
   239	        payload.password ||
   240	        payload.generatedPassword ||
   241	        payload.oneTimePassword ||
   242	        null,
   243	
   244	      initialBalance: toNumber(payload.initialBalance, 0),
   245	      currencyCode: payload.currencyCode || 'USD',
   246	
   247	      /**
   248	       * Important:
   249	       * Blockchain KYC creates a CUSTOMER wallet linked to an organization.
   250	       * Do not call createOrganizationWallet here because it generates ORG_* IDs,
   251	       * and enterprise tables require numeric customer_id.
   252	       */
   253	      walletType: 'CUSTOMER',
   254	      partyTypeCode: 7,
   255	
   256	      requestSource:
   257	        payload.requestSource ||
   258	        payload.request_source ||
   259	        'BLOCKCHAIN_KYC_MODULE',
   260	
   261	      sourceSystem:
   262	        payload.sourceSystem ||
   263	        payload.source_system ||
   264	        'BLOCKCHAIN_KYC_MODULE',
   265	
   266	      createdBy:
   267	        payload.createdBy ||
   268	        payload.created_by ||
   269	        'blockchain-kyc-module'
   270	    });
   271	
   272	    const updatedKycRequest = await markKycRequestWalletCreated(
   273	      kycRequest.request_id,
   274	      walletResult
   275	    );
   276	
   277	    return {
   278	      success: true,
   279	      message:
   280	        'Blockchain KYC customer wallet created successfully in enterprise tables, PostgreSQL wallet table, and Fabric ledger.',
   281	      kycRequest: updatedKycRequest,
   282	      walletCreationStatus: 'SUCCESS',
   283	      walletResult
   284	    };
   285	  } catch (error) {
   286	    const failedKycRequest = await markKycRequestWalletFailed(
   287	      kycRequest.request_id,
   288	      error
   289	    );
   290	
   291	    const finalError = new Error(
   292	      `KYC request was saved, but wallet creation failed: ${error.message}`
   293	    );
   294	
   295	    finalError.kycRequest = failedKycRequest;
   296	    finalError.originalError = error;
   297	    throw finalError;
   298	  }
   299	}
