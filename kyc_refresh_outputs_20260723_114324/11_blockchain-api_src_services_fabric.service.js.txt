   162	  async newSigner() {
   163	    const config = this.getConfig();
   164	
   165	    this.assertDirectoryExists(
   166	      config.keyDirectoryPath,
   167	      'FABRIC_KEY_DIRECTORY_PATH'
   168	    );
   169	
   170	    const files = fs
   171	      .readdirSync(config.keyDirectoryPath)
   172	      .filter((file) => !file.startsWith('.'));
   173	
   174	    if (!files || files.length === 0) {
   175	      throw new Error(
   176	        `No private key found inside FABRIC_KEY_DIRECTORY_PATH: ${config.keyDirectoryPath}`
   177	      );
   178	    }
   179	
   180	    const keyPath = path.join(config.keyDirectoryPath, files[0]);
   181	    const privateKeyPem = fs.readFileSync(keyPath);
   182	    const privateKey = crypto.createPrivateKey(privateKeyPem);
   183	
   184	    return signers.newPrivateKeySigner(privateKey);
   185	  }

   288	  async submitTransaction(functionName, args = [], context = {}) {
   289	    const startedAt = Date.now();
   290	    const config = this.getConfig();
   291	    const auditContext = this.normalizeContext(context);
   292	
   293	    try {
   294	      await auditService.log({
   295	        ...auditContext,
   296	        eventType: AUDIT_EVENT_TYPES.BLOCKCHAIN_SUBMIT_REQUEST,
   297	        eventCategory: AUDIT_EVENT_CATEGORY.BLOCKCHAIN,
   298	        eventStatus: AUDIT_EVENT_STATUS.PENDING,
   299	        blockchainFunction: functionName,
   300	        chaincodeName: config.chaincodeName,
   301	        channelName: config.channelName,
   302	        requestPayload: {
   303	          functionName,
   304	          args
   305	        },
   306	        serviceName: 'fabric.service'
   307	      });
   308	
   309	      const connection = await this.connect();
   310	
   311	      const normalizedArgs = this.normalizeArgs(args);
   312	      const endorsingOrganizations = this.getEndorsingOrganizations(
   313	        functionName,
   314	        config
   315	      );
   316	
   317	      let resultBuffer;
   318	      let transactionId = null;
   319	      let commitStatus = null;
   320	
   321	      console.log('[FABRIC SUBMIT CONFIG]', {
   322	        functionName,
   323	        channelName: config.channelName,
   324	        chaincodeName: config.chaincodeName,
   325	        peerEndpoint: config.peerEndpoint,
   326	        peerHostAlias: config.peerHostAlias,
   327	        endorsingOrganizations,
   328	        argsCount: normalizedArgs.length
   329	      });
   330	
   331	      if (
   332	        connection.contract &&
   333	        typeof connection.contract.newProposal === 'function' &&
   334	        endorsingOrganizations &&
   335	        endorsingOrganizations.length > 0
   336	      ) {
   337	        const proposal = connection.contract.newProposal(functionName, {
   338	          arguments: normalizedArgs,
   339	          endorsingOrganizations
   340	        });
   341	
   342	        const endorsedProposal = await proposal.endorse();
   343	
   344	        resultBuffer = endorsedProposal.getResult();
   345	
   346	        const submittedTransaction = await endorsedProposal.submit();
   347	
   348	        commitStatus = await submittedTransaction.getStatus();
   349	        transactionId = commitStatus.transactionId || null;
   350	
   351	        console.log('[FABRIC COMMIT STATUS]', {
   352	          functionName,
   353	          transactionId,
   354	          successful: commitStatus.successful,
   355	          code: commitStatus.code
   356	        });
   357	
   358	        if (!commitStatus.successful) {
   359	          throw new Error(
   360	            `Transaction commit failed with code ${commitStatus.code} for transaction ${commitStatus.transactionId}`
   361	          );
   362	        }
   363	      } else {
   364	        console.log('[FABRIC FALLBACK SUBMIT]', {
   365	          functionName,
   366	          reason: 'newProposal not available or no endorsing organizations'
   367	        });
   368	
   369	        resultBuffer = await connection.contract.submitTransaction(
   370	          functionName,
   371	          ...normalizedArgs
   372	        );
   373	      }
   374	
   375	      const parsedResult = this.parseBufferResult(resultBuffer);
   376	
   377	      const result = {
   378	        success: true,
   379	        type: 'submit',
   380	        channelName: config.channelName,
   381	        chaincodeName: config.chaincodeName,
   382	        functionName,
   383	        args,
   384	        transactionId,
   385	        txId: transactionId,
   386	        commitStatus,
   387	        data: parsedResult,
   388	        durationMs: Date.now() - startedAt
   389	      };
   390	
   391	      await auditService.log({
   392	        ...auditContext,
   393	        eventType: AUDIT_EVENT_TYPES.BLOCKCHAIN_SUBMIT_SUCCESS,
   394	        eventCategory: AUDIT_EVENT_CATEGORY.BLOCKCHAIN,
   395	        eventStatus: AUDIT_EVENT_STATUS.SUCCESS,
   396	        blockchainFunction: functionName,
   397	        chaincodeName: config.chaincodeName,
   398	        channelName: config.channelName,
   399	        responsePayload: result,
   400	        durationMs: Date.now() - startedAt,
   401	        serviceName: 'fabric.service'
   402	      });
   403	
   404	      return result;
   405	    } catch (error) {
   406	      await auditService.log({
   407	        ...auditContext,
   408	        eventType: AUDIT_EVENT_TYPES.BLOCKCHAIN_SUBMIT_FAILED,
   409	        eventCategory: AUDIT_EVENT_CATEGORY.BLOCKCHAIN,
   410	        eventStatus: AUDIT_EVENT_STATUS.FAILED,
   411	        blockchainFunction: functionName,
   412	        chaincodeName: config.chaincodeName,
   413	        channelName: config.channelName,
   414	        errorCode: error.code || 'FABRIC_SUBMIT_ERROR',
   415	        errorMessage: error.message,
   416	        errorStack: error.stack,
   417	        requestPayload: {
   418	          functionName,
   419	          args
   420	        },
   421	        durationMs: Date.now() - startedAt,
   422	        serviceName: 'fabric.service'
   423	      });
   424	
   425	      throw error;
   426	    }
   427	  }
